cmake_minimum_required(VERSION 3.20)
project(aphid_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(aphid_bridge SHARED bridge.cpp)
target_include_directories(aphid_bridge PRIVATE
    ${LADYBUG_SOURCE}/src/include ${LADYBUG_BUILD}/src/include ${LADYBUG_SOURCE}/third_party/yyjson/src)
target_link_libraries(aphid_bridge PRIVATE ${LADYBUG_BUILD}/src/${CMAKE_SHARED_LIBRARY_PREFIX}lbug${CMAKE_SHARED_LIBRARY_SUFFIX} Threads::Threads)
option(APHID_BUILD_TESTS "Build isolated lifecycle fault-injection test" OFF)
if(APHID_BUILD_TESTS)
    add_executable(extension_concurrency tests/extension_concurrency.cpp)
    target_include_directories(extension_concurrency PRIVATE
        ${LADYBUG_SOURCE}/src/include ${LADYBUG_BUILD}/src/include)
    target_link_libraries(extension_concurrency PRIVATE ${LADYBUG_BUILD}/src/${CMAKE_SHARED_LIBRARY_PREFIX}lbug${CMAKE_SHARED_LIBRARY_SUFFIX} Threads::Threads)
    if(APHID_TEST_MAX_DB_SIZE)
        target_compile_definitions(extension_concurrency PRIVATE APHID_TEST_MAX_DB_SIZE=${APHID_TEST_MAX_DB_SIZE})
    endif()
    add_executable(lifecycle tests/lifecycle.cpp)
    if(APHID_TEST_MAX_DB_SIZE)
        target_compile_definitions(lifecycle PRIVATE APHID_TEST_MAX_DB_SIZE=${APHID_TEST_MAX_DB_SIZE})
    endif()
    target_include_directories(lifecycle PRIVATE
        ${LADYBUG_SOURCE}/src/include ${LADYBUG_BUILD}/src/include ${LADYBUG_SOURCE}/third_party/yyjson/src)
    target_link_libraries(lifecycle PRIVATE ${LADYBUG_BUILD}/src/${CMAKE_SHARED_LIBRARY_PREFIX}lbug${CMAKE_SHARED_LIBRARY_SUFFIX} Threads::Threads)
endif()
